"""Parsers for the three Abramus Digital report types.

- Relatório Geral: CSV (sometimes shipped inside a ZIP, one file per editora).
  Delimiter: comma. Decimal: comma (Brazilian). Columns include:
    Competência, Período, Cliente, Editora, Território, Fonte, Formato,
    Pagamento, Id Obra, ISWC, Título da Obra, Autores, Interprete, ISRC,
    Unidades, % Controle, Val. Liquido

- Relatório Editora: XLSX with columns:
    PERIODO, CLIENTE, TITULAR, NOME, CAT, ID_OBRA, OBRA, FONTE, PAGAMENTO,
    AUTORES, UNIDADES, PERC_TOTAL, PERC_RECEBIDO, LIQUIDO, PERC_EDITORA,
    VL_EDITORA, ADMINISTRACAO_CLIENTE, VL_TAXA

- Relatório Autor: XLSX with columns:
    COMPETENCIA, CLIENTE, TITULAR, NOME, ID_OBRA, OBRA, TIPO_UTILIZACAO,
    TERRITORIO, FONTE, FORMATO, PERIODO, AUTORES, UNIDADES, PERC_TOTAL,
    PERC_RECEBIDO, LIQUIDO, PERC_AUTOR, VL_AUTOR, VL_EDITORA
"""
import io
import math
import zipfile
import pandas as pd
import unicodedata


def _norm(s: str) -> str:
    if s is None:
        return ""
    s = str(s).strip().upper()
    s = unicodedata.normalize("NFKD", s).encode("ASCII", "ignore").decode("ASCII")
    return " ".join(s.split())


def _to_float_br(v) -> float:
    if v is None:
        return 0.0
    if isinstance(v, (int, float)):
        if isinstance(v, float) and math.isnan(v):
            return 0.0
        return float(v)
    s = str(v).strip()
    if not s:
        return 0.0
    # Brazilian decimal: "1.234,56" -> "1234.56"
    s = s.replace(".", "").replace(",", ".") if "," in s else s
    try:
        return float(s)
    except Exception:
        return 0.0


def _clean_value(v):
    """Return a JSON/Mongo friendly value (no NaN, no numpy types)."""
    if v is None:
        return None
    if isinstance(v, float):
        if math.isnan(v):
            return None
        return float(v)
    if hasattr(v, "item"):
        try:
            return v.item()
        except Exception:
            return str(v)
    if isinstance(v, (int, str, bool)):
        return v
    return str(v)


def parse_general_csv(file_bytes: bytes, filename: str) -> list[dict]:
    """Parse one Relatório Geral CSV file.
    Returns a list of normalized dict rows."""
    try:
        text = file_bytes.decode("utf-8-sig")
    except UnicodeDecodeError:
        text = file_bytes.decode("latin-1")
    df = pd.read_csv(io.StringIO(text), sep=",", dtype=str, keep_default_na=False)
    rows = []
    for _, r in df.iterrows():
        rows.append(
            {
                "competencia": r.get("Competência", ""),
                "periodo": r.get("Período", ""),
                "cliente": r.get("Cliente", ""),
                "editora_nome": r.get("Editora", ""),
                "editora_norm": _norm(r.get("Editora", "")),
                "territorio": r.get("Território", ""),
                "fonte": r.get("Fonte", ""),
                "formato": r.get("Formato", ""),
                "pagamento": r.get("Pagamento", ""),
                "id_obra": r.get("Id Obra", ""),
                "iswc": r.get("ISWC", ""),
                "titulo_obra": r.get("Título da Obra", ""),
                "autores": r.get("Autores", ""),
                "interprete": r.get("Interprete", ""),
                "isrc": r.get("ISRC", ""),
                "unidades": int(_to_float_br(r.get("Unidades", 0))),
                "perc_controle": _to_float_br(r.get("% Controle", 0)),
                "val_liquido": _to_float_br(r.get("Val. Liquido", 0)),
                "source_file": filename,
            }
        )
    return rows


def parse_general_zip_or_csv(file_bytes: bytes, filename: str) -> list[dict]:
    """Handle either a single CSV or a ZIP of CSVs."""
    all_rows = []
    if filename.lower().endswith(".zip") or file_bytes[:2] == b"PK":
        zf = zipfile.ZipFile(io.BytesIO(file_bytes))
        for name in zf.namelist():
            if name.endswith("/") or not name.lower().endswith(".csv"):
                continue
            with zf.open(name) as fh:
                data = fh.read()
            all_rows.extend(parse_general_csv(data, name))
    else:
        all_rows.extend(parse_general_csv(file_bytes, filename))
    return all_rows

def parse_bulk_fees_csv(file_bytes: bytes) -> list[dict]:
    """Parse a CSV with columns: nome,taxa_admin (percent 0-100). Optional column: editora (for autores)."""
    try:
        text = file_bytes.decode("utf-8-sig")
    except UnicodeDecodeError:
        text = file_bytes.decode("latin-1")
    sep = "," if text.count(",") >= text.count(";") else ";"
    df = pd.read_csv(io.StringIO(text), sep=sep, dtype=str, keep_default_na=False)
    df.columns = [c.strip().lower() for c in df.columns]
    out = []
    for _, r in df.iterrows():
        out.append(
            {
                "nome": (r.get("nome") or r.get("name") or "").strip(),
                "taxa_admin": _to_float_br(r.get("taxa_admin") or r.get("taxa") or r.get("fee") or 0),
                "editora": (r.get("editora") or "").strip(),
                "cnpj": (r.get("cnpj") or "").strip(),
                "email": (r.get("email") or "").strip(),
            }
        )
    return out

def _env(name: str, default: str | None = None) -> str | None:

_SHORTENERS = ("bit.ly", "tinyurl.com", "t.co", "is.gd", "cutt.ly", "goo.gl", "rebrand.ly")
_CRED_ASK = (
    "reply with your password", "reply with the code", "send your password", "cvv",
    "send us your password", "enter your password below", "confirm your card number",
    "your full card number", "seed phrase", "recovery phrase", "verify your card",
    "social security number", "confirm your bank details",
)
_HOSTISH = re.compile(r"\b(?:https?://)?((?:[a-z0-9-]+\.)+[a-z]{2,})", re.I)


def _host_ok(host: str) -> bool:
    if not host or "xn--" in host:
        return False
    try:
        ipaddress.ip_address(host)
        return False
    except ValueError:
        pass
    return not any(host == s or host.endswith("." + s) for s in _SHORTENERS)


def _same_site(shown: str, real: str) -> bool:
    return shown == real or real.endswith("." + shown) or shown.endswith("." + real)


class _EmailScan(HTMLParser):
    def __init__(self):
        super().__init__()
        self.tags, self.urls, self.anchors = set(), [], []
        self._href, self._text = None, []

    def handle_starttag(self, tag, attrs):
        self.tags.add(tag.lower())
        self.urls += [v for k, v in attrs if k.lower() in ("href", "src") and v]
        if tag.lower() == "a":
            self._href = dict((k.lower(), v) for k, v in attrs).get("href")
            self._text = []

    def handle_data(self, data):
        if self._href is not None:
            self._text.append(data)

    def handle_endtag(self, tag):
        if tag.lower() == "a" and self._href is not None:
            self.anchors.append((self._href, "".join(self._text)))
            self._href, self._text = None, []


def _assert_safe_email(subject: str, html: str) -> None:
    scan = _EmailScan()
    scan.feed(html)
    if scan.tags & {"form", "input", "textarea", "select"}:
        raise ValueError("No forms or input fields in email (G2)")
    body = f"{subject}\n{html}".lower()
    for p in _CRED_ASK:
        if p in body:
            raise ValueError(f"Email asks the recipient for credentials: {p!r} (G2)")
    for url in scan.urls:
        low = url.strip().lower()
        if low.startswith(("mailto:", "tel:", "cid:", "#")):
            continue
        if not low.startswith("https://"):
            raise ValueError(f"Email links/assets must be absolute https: {url!r} (G3)")
        host = urlparse(low).hostname or ""
        if not _host_ok(host) or urlparse(low).username is not None:
            raise ValueError(f"Shortened, numeric-host or credential-bearing URL: {url!r} (G3)")
    for href, text in scan.anchors:
        real = urlparse(href.strip().lower()).hostname or ""
        if not real:
            continue
        for m in _HOSTISH.finditer(text):
            if not _same_site(m.group(1).lower(), real):
                raise ValueError(f"Anchor text {m.group(1)!r} != real link host {real!r} (G3)")


async def send_email(*, to: str, subject: str, html: str, reply_to: str | None = None) -> str | None:
    key = _env("EMERGENT_EMAIL_KEY")
    from_name = _env("EMAIL_FROM_NAME", "Acorde Music Rights")
    if not key:
        raise RuntimeError("EMERGENT_EMAIL_KEY not configured")
    _assert_safe_email(subject, html)
    payload = {"to": [to], "subject": subject, "html": html, "from_name": from_name}
    reply = reply_to or _env("EMAIL_REPLY_TO")
    if reply:
        payload["contact_email"] = reply
    async with httpx.AsyncClient(timeout=30) as client:
        resp = await client.post(
            f"{EMAIL_BASE_URL}/api/v1/email/send",
            headers={"X-Email-Key": key},
            json=payload,
        )
    if resp.status_code >= 400:
        logger.error("Email send failed %s %s", resp.status_code, resp.text)
        raise RuntimeError(f"Email send failed: {resp.status_code}")
    try:
        return resp.json().get("id")
    except Exception:
        return None

def _p(text, right=False):
    """Wrap value as Paragraph so it word-wraps inside the cell."""
    if text is None:
        text = ""
    text = str(text).replace("&", "&amp;").replace("<", "&lt;").replace(">", "&gt;")
    return Paragraph(text, _cell_style_right() if right else _cell_style())

def _money(v) -> str:
    try:
        v = float(v)
    except Exception:
        v = 0.0
    s = f"{v:,.2f}"
    # BR format: thousands "." decimal ","
    s = s.replace(",", "X").replace(".", ",").replace("X", ".")
    return f"R$ {s}"


def _pct(v) -> str:
    try:
        v = float(v)
    except Exception:
        v = 0.0
    return f"{v:.2f}%".replace(".", ",")

def generate_editora_pdf(
    editora_name: str,
    cnpj: str,
    competencia: str,
    admin_fee_percent: float,
    rows: list[dict],
    totals: dict,
) -> bytes:
    """rows items should contain: fonte, obra, autores, unidades, vl_editora_bruto,
    taxa_admin_valor, vl_editora_liquido."""
    buf = io.BytesIO()
    doc = SimpleDocTemplate(
        buf,
        pagesize=landscape(A4),
        topMargin=35 * mm,
        bottomMargin=15 * mm,
        leftMargin=12 * mm,
        rightMargin=12 * mm,
    )
    styles = _styles()
    story = []

    title = f"RELATÓRIO EDITORA"
    subtitle = f"{editora_name}"
    story.append(Paragraph(f"<b>Relatório Analítico • Editora</b>", styles["AcordeH1"]))
    story.append(Paragraph(f"{editora_name}", styles["AcordeBody"]))
    if cnpj:
        story.append(Paragraph(f"CNPJ: {cnpj}", styles["AcordeMuted"]))
    story.append(Paragraph(f"Competência: <b>{competencia or '—'}</b>", styles["AcordeMuted"]))
    story.append(Spacer(1, 8))

    summary = [
        ["VALOR BRUTO EDITORA", _money(totals.get("bruto", 0))],
        [f"TAXA ADMINISTRAÇÃO ({_pct(admin_fee_percent)})", _money(totals.get("taxa", 0))],
        ["VALOR LÍQUIDO A REPASSAR", _money(totals.get("liquido", 0))],
    ]
    story.append(_summary_table(summary))
    story.append(Spacer(1, 10))

    story.append(Paragraph("Detalhamento por obra", styles["AcordeH2"]))

    headers = [
        "Fonte",
        "Obra",
        "Autores",
        "Unid.",
        "% Editora",
        "Vl. Bruto",
        "Taxa Admin.",
        "Vl. Líquido",
    ]
    body = []
    for r in rows:
        body.append(
            [
                _p(r.get("fonte", "")),
                _p(r.get("obra", "")),
                _p(r.get("autores", "")),
                f"{int(r.get('unidades', 0)):,}".replace(",", "."),
                _pct(float(r.get("perc_editora", 0)) * 100 if float(r.get("perc_editora", 0)) <= 1 else float(r.get("perc_editora", 0))),
                _money(r.get("vl_editora_bruto", 0)),
                _money(r.get("taxa_admin_valor", 0)),
                _money(r.get("vl_editora_liquido", 0)),
            ]
        )
    body.append(
        [
            "",
            "",
            "TOTAIS",
            "",
            "",
            _money(totals.get("bruto", 0)),
            _money(totals.get("taxa", 0)),
            _money(totals.get("liquido", 0)),
        ]
    )
    col_widths = [30 * mm, 55 * mm, 65 * mm, 15 * mm, 18 * mm, 24 * mm, 24 * mm, 25 * mm]
    tbl = _detail_table(headers, body, col_widths)
    tbl.setStyle(
        TableStyle(
            [
                ("BACKGROUND", (0, -1), (-1, -1), ACORDE_BLACK),
                ("TEXTCOLOR", (0, -1), (-1, -1), ACORDE_YELLOW),
                ("FONTNAME", (0, -1), (-1, -1), "Helvetica-Bold"),
                ("ALIGN", (3, 1), (-1, -1), "RIGHT"),
            ]
        )
    )
    story.append(tbl)

    def _on_page(canvas, d):
        _header_footer(canvas, d, title, subtitle)

    doc.build(story, onFirstPage=_on_page, onLaterPages=_on_page)
    return buf.getvalue()

@api.post("/auth/login")
async def login(data: LoginIn, response: Response):
    user = await db.users.find_one({"email": data.email.lower()})
    if not user or not verify_password(data.password, user["password_hash"]):
        raise HTTPException(status_code=401, detail="Credenciais inválidas")
    user_id = str(user["_id"])
    token = create_access_token(user_id, user["email"], user["role"])
    response.set_cookie(
        key="access_token",
        value=token,
        httponly=True,
        secure=True,
        samesite="none",
        max_age=12 * 3600,
        path="/",
    )
    return {
        "token": token,
        "user": {
            "id": user_id,
            "email": user["email"],
            "name": user.get("name", ""),
            "role": user["role"],
            "editora_id": user.get("editora_id"),
        },
    }

                    "admin_fee_percent": r["taxa_admin"],
                    "needs_review": False,
                    "created_at": datetime.now(timezone.utc).isoformat(),
                }
            )
            created += 1
    return {"created": created, "updated": updated, "total": created + updated}

async def _store_rows(rows: list[dict], upload_id: str, rtype: str):
    if not rows:
        return 0
    for r in rows:
        r["upload_id"] = upload_id
        r["type"] = rtype
    await db.royalty_rows.insert_many(rows)
    return len(rows)


async def _auto_discover(rtype: str, rows: list[dict]) -> dict:
    """After an upload, auto-create stub editoras/autores for every unique
    titular/nome found in the data. Existing records are preserved untouched.
    Stubs have admin_fee_percent=0 and needs_review=True so they appear as
    'pending' until an admin configures the taxa."""
    now = datetime.now(timezone.utc).isoformat()
    new_editoras, new_autores = 0, 0

    if rtype in ("geral", "editora"):
        # unique editoras
        seen = {}
        for r in rows:
            name = (r.get("editora_nome") or r.get("titular") or "").strip()
            if not name:
                continue
            key = _norm(name)
            if key and key not in seen:
                seen[key] = name
        for name_norm, name in seen.items():
            existing = await db.editoras.find_one({"name_norm": name_norm})
            if not existing:
                await db.editoras.insert_one({
                    "name": name,
                    "name_norm": name_norm,
                    "cnpj": "",
                    "email": "",
                    "admin_fee_percent": 0.0,
                    "needs_review": True,
                    "auto_created": True,
                    "created_at": now,
                })
                new_editoras += 1

    if rtype == "autor":
        # unique (titular, nome) pairs, resolving editora_id by titular_norm
        pairs = {}
        for r in rows:
            name = (r.get("nome") or "").strip()
            titular = (r.get("titular") or "").strip()
            if not name:
                continue
            key = (_norm(name), _norm(titular))
            if key not in pairs:
                pairs[key] = (name, titular)
        # make sure the editoras exist first (same auto-stub logic)
        unique_editoras = {}
        for (_, t_norm), (_, titular) in pairs.items():
            if t_norm and t_norm not in unique_editoras:
                unique_editoras[t_norm] = titular
        for t_norm, titular in unique_editoras.items():
            existing = await db.editoras.find_one({"name_norm": t_norm})
            if not existing:
                await db.editoras.insert_one({
                    "name": titular,
                    "name_norm": t_norm,
                    "cnpj": "",
                    "email": "",
                    "admin_fee_percent": 0.0,
                    "needs_review": True,
                    "auto_created": True,
                    "created_at": now,
                })
                new_editoras += 1
        # now autores
        for (n_norm, t_norm), (name, titular) in pairs.items():
            editora_id = None
            if t_norm:
                ed = await db.editoras.find_one({"name_norm": t_norm})
                if ed:
                    editora_id = str(ed["_id"])
            existing = await db.autores.find_one({"name_norm": n_norm, "editora_id": editora_id})
            if not existing:
                try:
                    await db.autores.insert_one({
                        "name": name,
                        "name_norm": n_norm,
                        "editora_id": editora_id,
                        "email": "",
                        "admin_fee_percent": 0.0,
                        "needs_review": True,
                        "auto_created": True,
                        "created_at": now,
                    })
                    new_autores += 1
                except Exception:
                    pass

    return {"new_editoras": new_editoras, "new_autores": new_autores}


@api.post("/uploads/{report_type}")
async def upload_report(report_type: str, request: Request, file: UploadFile = File(...)):
    user = await get_current_user(request)
    require_admin(user)
    if report_type not in ("geral", "editora", "autor"):
        raise HTTPException(400, "Tipo inválido")
    content = await file.read()
    filename = file.filename or ""
    if report_type == "geral":
        rows = parse_general_zip_or_csv(content, filename)
    elif report_type == "editora":
        rows = parse_editora_xlsx(content, filename)
    else:
        rows = parse_autor_xlsx(content, filename)

    competencias = sorted({(r.get("competencia") or r.get("periodo") or "").strip() for r in rows if (r.get("competencia") or r.get("periodo"))})
    competencia = competencias[0] if competencias else ""

    upload_doc = {
        "filename": filename,
        "type": report_type,
        "competencia": competencia,
        "rows_count": len(rows),
        "uploaded_at": datetime.now(timezone.utc).isoformat(),
        "uploaded_by": user["email"],
    }
    res = await db.uploads.insert_one(upload_doc)
    upload_id = str(res.inserted_id)
    stored = await _store_rows(rows, upload_id, report_type)
    discovery = await _auto_discover(report_type, rows)
    # Update upload with discovery results
    await db.uploads.update_one({"_id": res.inserted_id}, {"$set": {"discovery": discovery}})

    pending_editoras = await db.editoras.count_documents({"needs_review": True})
    pending_autores = await db.autores.count_documents({"needs_review": True})

    return {
        "upload_id": upload_id,
        "type": report_type,
        "rows": stored,
        "competencia": competencia,
        "discovery": discovery,
        "pending_editoras": pending_editoras,
        "pending_autores": pending_autores,
    }

async def _editora_report(editora_id: str, competencia: Optional[str]):
    editora = await db.editoras.find_one({"_id": ObjectId(editora_id)})
    if not editora:
        raise HTTPException(404, "Editora não encontrada")
    q = {"type": "editora", "titular_norm": editora["name_norm"]}
    if competencia:
        q["$or"] = [{"competencia": competencia}, {"periodo": competencia}]
    rows_raw = await db.royalty_rows.find(q).to_list(100000)
    fee_percent = float(editora.get("admin_fee_percent", 0) or 0)
    fee_mult = fee_percent / 100.0
    rows_out = []
    total_bruto = 0.0
    total_taxa = 0.0
    for r in rows_raw:
        bruto = float(r.get("vl_editora", 0) or 0)
        taxa = round(bruto * fee_mult, 6)
        liquido = round(bruto - taxa, 6)
        rows_out.append(
            {
                "fonte": r.get("fonte", ""),
                "obra": r.get("obra", ""),
                "autores": r.get("autores", ""),
                "unidades": r.get("unidades", 0),
                "perc_editora": r.get("perc_editora", 0),
                "pagamento": r.get("pagamento", ""),
                "vl_editora_bruto": bruto,
                "taxa_admin_valor": taxa,
                "vl_editora_liquido": liquido,
                "competencia": r.get("competencia") or r.get("periodo", ""),
            }
        )
        total_bruto += bruto
        total_taxa += taxa
    return {
        "editora": {
            "id": str(editora["_id"]),
            "name": editora["name"],
            "cnpj": editora.get("cnpj", ""),
            "admin_fee_percent": fee_percent,
        },
        "competencia": competencia or "",
        "rows": rows_out,
        "totals": {
            "bruto": round(total_bruto, 2),
            "taxa": round(total_taxa, 2),
            "liquido": round(total_bruto - total_taxa, 2),
        },
    }


async def _autor_report(autor_id: str, competencia: Optional[str]):
    autor = await db.autores.find_one({"_id": ObjectId(autor_id)})
    if not autor:
        raise HTTPException(404, "Autor não encontrado")
    editora_name = ""
    if autor.get("editora_id"):
        e = await db.editoras.find_one({"_id": ObjectId(autor["editora_id"])})
        if e:
            editora_name = e["name"]
    q = {"type": "autor", "nome_norm": autor["name_norm"]}
    if competencia:
        q["$or"] = [{"competencia": competencia}, {"periodo": competencia}]
    rows_raw = await db.royalty_rows.find(q).to_list(100000)
    fee_percent = float(autor.get("admin_fee_percent", 0) or 0)
    fee_mult = fee_percent / 100.0
    rows_out = []
    total_bruto = 0.0
    total_liquido = 0.0
    total_units = 0
    obras = set()
    for r in rows_raw:
        bruto = float(r.get("vl_autor", 0) or 0)
        liquido = round(bruto * (1 - fee_mult), 6)
        rows_out.append(
            {
                "fonte": r.get("fonte", ""),
                "formato": r.get("formato", ""),
                "territorio": r.get("territorio", ""),
                "obra": r.get("obra", ""),
                "periodo": r.get("periodo") or r.get("competencia", ""),
                "unidades": r.get("unidades", 0),
                "perc_autor": r.get("perc_autor", 0),
                "vl_bruto": bruto,  # hidden on PDF, admin-only
                "vl_liquido": liquido,
                "autores": r.get("autores", ""),
            }
        )
        total_bruto += bruto
        total_liquido += liquido
        total_units += int(r.get("unidades", 0) or 0)
        obras.add(r.get("obra", ""))
    return {
        "autor": {
            "id": str(autor["_id"]),
            "name": autor["name"],
            "admin_fee_percent": fee_percent,
            "editora_id": autor.get("editora_id"),
            "editora_name": editora_name,
        },
        "competencia": competencia or "",
        "rows": rows_out,
        "totals": {
            "bruto": round(total_bruto, 2),
            "liquido": round(total_liquido, 2),
            "unidades": total_units,
            "obras": len({o for o in obras if o}),
        },
    }

@api.get("/dashboard/summary")
async def dashboard_summary(request: Request, competencia: Optional[str] = None):
    user = await get_current_user(request)
    is_admin = user["role"] == "admin"

    match_autor = {"type": "autor"}
    match_editora = {"type": "editora"}
    match_geral = {"type": "geral"}
    if competencia:
        match_autor["$or"] = [{"competencia": competencia}, {"periodo": competencia}]
        match_editora["$or"] = [{"competencia": competencia}, {"periodo": competencia}]
        match_geral["$or"] = [{"competencia": competencia}, {"periodo": competencia}]

    if not is_admin:
        # scope to editora
        editora = await db.editoras.find_one({"_id": ObjectId(user["editora_id"])})
        if not editora:
            raise HTTPException(404, "Editora não encontrada")
        match_autor["titular_norm"] = editora["name_norm"]
        match_editora["titular_norm"] = editora["name_norm"]
        match_geral["editora_norm"] = editora["name_norm"]

    pipeline_sum = lambda match, field: [
        {"$match": match},
        {"$group": {"_id": None, "total": {"$sum": f"${field}"}, "count": {"$sum": 1}}},
    ]

    total_autor = await db.royalty_rows.aggregate(pipeline_sum(match_autor, "vl_autor")).to_list(1)
    total_editora = await db.royalty_rows.aggregate(pipeline_sum(match_editora, "vl_editora")).to_list(1)
    total_geral = await db.royalty_rows.aggregate(pipeline_sum(match_geral, "val_liquido")).to_list(1)

    def _val(x):
        return (x[0]["total"] if x else 0) or 0

    def _cnt(x):
        return (x[0]["count"] if x else 0) or 0

    # breakdown by fonte (autor rows)
    por_fonte = await db.royalty_rows.aggregate(
        [
            {"$match": match_autor},
            {"$group": {"_id": "$fonte", "total": {"$sum": "$vl_autor"}, "count": {"$sum": 1}}},
            {"$sort": {"total": -1}},
            {"$limit": 10},
        ]
    ).to_list(10)

    # Top editoras (admin only)
    top_editoras = []
    if is_admin:
        top_editoras = await db.royalty_rows.aggregate(
            [
                {"$match": match_editora},
                {"$group": {"_id": "$titular", "total": {"$sum": "$vl_editora"}}},
                {"$sort": {"total": -1}},
                {"$limit": 10},
            ]
        ).to_list(10)

    editoras_count = await db.editoras.count_documents({})
    autores_count = await db.autores.count_documents({"editora_id": user["editora_id"]} if not is_admin else {})

    return {
        "role": user["role"],
        "competencia": competencia or "",
        "totals": {
            "valor_total_geral": round(_val(total_geral), 2),
            "valor_total_editora": round(_val(total_editora), 2),
            "valor_total_autor": round(_val(total_autor), 2),
            "linhas_autor": _cnt(total_autor),
            "linhas_editora": _cnt(total_editora),
            "linhas_geral": _cnt(total_geral),
            "editoras_cadastradas": editoras_count,
            "autores_cadastrados": autores_count,
        },
        "por_fonte": [{"fonte": p["_id"] or "—", "total": round(p["total"], 2), "count": p["count"]} for p in por_fonte],
        "top_editoras": [{"titular": t["_id"] or "—", "total": round(t["total"], 2)} for t in top_editoras],
    }

async def _check_repasse_scope(user, scope, scope_id):
    if user["role"] != "editora":

            raise HTTPException(403, "Sem acesso a este repasse")

async def upsert_repasse(payload: RepasseIn, request: Request):
    user = await get_current_user(request)
    require_admin(user)
    if payload.scope not in ("editora", "autor"):
        raise HTTPException(400, "scope inválido")
    if payload.status not in ("pendente", "pago"):
        raise HTTPException(400, "status inválido")
    now = datetime.now(timezone.utc).isoformat()
    event = {
        "status": payload.status,
        "valor": float(payload.valor or 0),
        "observacao": payload.observacao or "",
        "data_pagamento": payload.data_pagamento,
        "at": now,
        "by": user["email"],
    }
    existing = await db.repasses.find_one(
        {"scope": payload.scope, "scope_id": payload.scope_id, "competencia": payload.competencia}
    )
    if existing:
        new_history = (existing.get("history") or []) + [event]
        await db.repasses.update_one(
            {"_id": existing["_id"]},
            {"$set": {
                "status": payload.status,
                "valor": event["valor"],
                "observacao": event["observacao"],
                "data_pagamento": payload.data_pagamento,
                "updated_at": now,
                "history": new_history[-50:],
            }},
        )
        doc = await db.repasses.find_one({"_id": existing["_id"]})
    else:
        doc = {
            "scope": payload.scope,
            "scope_id": payload.scope_id,
            "competencia": payload.competencia,
            "status": payload.status,
            "valor": event["valor"],
            "observacao": event["observacao"],
            "data_pagamento": payload.data_pagamento,
            "created_at": now,
            "updated_at": now,
            "history": [event],
        }
        res = await db.repasses.insert_one(doc)
        doc["_id"] = res.inserted_id
    return _dump_repasse(doc)

def _safe_filename(s: str) -> str:
    return re.sub(r"[^A-Za-z0-9_\-]+", "_", (s or "sem_nome"))[:80]


@api.get("/reports/bulk/{scope}/pdf")
async def bulk_pdf_zip(scope: str, request: Request, competencia: Optional[str] = None):
    """Return a ZIP containing all PDFs for the given scope (editora|autor|both)

            editoras = await db.editoras.find({}).to_list(5000)
            for e in editoras:
                data = await _editora_report(str(e["_id"]), competencia)
                if not data["rows"]:
                    continue
                pdf = generate_editora_pdf(
                    editora_name=data["editora"]["name"],
                    cnpj=data["editora"].get("cnpj", ""),
                    competencia=data["competencia"],
                    admin_fee_percent=data["editora"]["admin_fee_percent"],
                    rows=data["rows"],
                    totals=data["totals"],
                )
                fname = f"Editoras/Acorde_Editora_{_safe_filename(data['editora']['name'])}.pdf"
                zf.writestr(fname, pdf)
                files_count += 1

        if scope in ("autor", "both"):
            autores = await db.autores.find({}).to_list(10000)
            for a in autores:
                data = await _autor_report(str(a["_id"]), competencia)
                if not data["rows"]:

                    autor_name=data["autor"]["name"],
                    editora_name=data["autor"]["editora_name"],
                    competencia=data["competencia"],
                    rows=data["rows"],
                    totals=data["totals"],
                )
                fname = f"Autores/Acorde_Autor_{_safe_filename(data['autor']['name'])}.pdf"
                zf.writestr(fname, pdf)
                files_count += 1

    buf.seek(0)
    if files_count == 0:
        raise HTTPException(404, "Nenhum relatório encontrado para essa competência")
    tag = _safe_filename(competencia or "todas")
    return StreamingResponse(
        buf,
        media_type="application/zip",
        headers={
            "Content-Disposition": f'attachment; filename="Acorde_Relatorios_{scope}_{tag}.zip"',
            "X-Files-Count": str(files_count),
        },

@api.post("/discover/sync")
async def discover_sync(request: Request):
    """Scan all royalty rows already in DB and auto-create any missing
    editoras/autores. Useful when data was imported before auto-discovery
    existed, or when a report was re-imported after record deletion.
    Returns totals and lists of pending (needs_review) items."""
    user = await get_current_user(request)
    require_admin(user)

    now = datetime.now(timezone.utc).isoformat()

    # Editoras: union of titular_norm (editora/autor rows) + editora_norm (geral)
    editoras_norms = set()
    editoras_names = {}
    async for r in db.royalty_rows.find({"type": {"$in": ["editora", "autor"]}}, {"titular": 1, "titular_norm": 1}):
        n = r.get("titular_norm")
        if n and n not in editoras_norms:
            editoras_norms.add(n)
            editoras_names[n] = r.get("titular", "")
    async for r in db.royalty_rows.find({"type": "geral"}, {"editora_nome": 1, "editora_norm": 1}):
        n = r.get("editora_norm")
        if n and n not in editoras_norms:
            editoras_norms.add(n)
            editoras_names[n] = r.get("editora_nome", "")

    new_editoras = 0
    for norm, name in editoras_names.items():
        existing = await db.editoras.find_one({"name_norm": norm})
        if not existing and name:
            await db.editoras.insert_one({
                "name": name,
                "name_norm": norm,
                "cnpj": "",
                "email": "",
                "admin_fee_percent": 0.0,
                "needs_review": True,
                "auto_created": True,
                "created_at": now,
            })
            new_editoras += 1

    # Autores: unique (nome_norm, titular_norm) from autor rows
    pairs = {}
    async for r in db.royalty_rows.find({"type": "autor"}, {"nome": 1, "nome_norm": 1, "titular": 1, "titular_norm": 1}):
        n, t = r.get("nome_norm"), r.get("titular_norm")
        if n and (n, t) not in pairs:
            pairs[(n, t)] = (r.get("nome", ""), r.get("titular", ""))

    new_autores = 0
    for (n_norm, t_norm), (name, titular) in pairs.items():
        editora_id = None
        if t_norm:
            ed = await db.editoras.find_one({"name_norm": t_norm})
            if ed:
                editora_id = str(ed["_id"])
        existing = await db.autores.find_one({"name_norm": n_norm, "editora_id": editora_id})
        if not existing:
            try:
                await db.autores.insert_one({
                    "name": name,
                    "name_norm": n_norm,
                    "editora_id": editora_id,
                    "email": "",
                    "admin_fee_percent": 0.0,
                    "needs_review": True,
                    "auto_created": True,
                    "created_at": now,
                })
                new_autores += 1
            except Exception:
                pass

    pending_editoras = await db.editoras.count_documents({"needs_review": True})
    pending_autores = await db.autores.count_documents({"needs_review": True})
    return {
        "new_editoras": new_editoras,
        "new_autores": new_autores,
        "pending_editoras": pending_editoras,
        "pending_autores": pending_autores,
    }


def _csv_response(rows: list[list[str]], filename: str) -> StreamingResponse:
    buf = io.StringIO()
    import csv as _csv
    writer = _csv.writer(buf, delimiter=",", quotechar='"', quoting=_csv.QUOTE_MINIMAL)
    for row in rows:
        writer.writerow(row)
    data = buf.getvalue().encode("utf-8-sig")  # BOM for Excel BR
    return StreamingResponse(
        io.BytesIO(data),
        media_type="text/csv; charset=utf-8",
        headers={"Content-Disposition": f'attachment; filename="{filename}"'},
    )

"""Authentication helpers: bcrypt + JWT + current user dependency."""
import os
import bcrypt
import jwt
from datetime import datetime, timezone, timedelta
from fastapi import HTTPException, Request, Depends
from bson import ObjectId


JWT_ALGORITHM = "HS256"
ACCESS_TOKEN_HOURS = 12


def hash_password(password: str) -> str:
    salt = bcrypt.gensalt()
    hashed = bcrypt.hashpw(password.encode("utf-8"), salt)
    return hashed.decode("utf-8")


def verify_password(plain: str, hashed: str) -> bool:
    try:
        return bcrypt.checkpw(plain.encode("utf-8"), hashed.encode("utf-8"))
    except Exception:
        return False


def get_jwt_secret() -> str:
    return os.environ["JWT_SECRET"]


def create_access_token(user_id: str, email: str, role: str) -> str:
    payload = {
        "sub": user_id,
        "email": email,
        "role": role,
        "exp": datetime.now(timezone.utc) + timedelta(hours=ACCESS_TOKEN_HOURS),
        "type": "access",
    }
    return jwt.encode(payload, get_jwt_secret(), algorithm=JWT_ALGORITHM)


def decode_token(token: str) -> dict:
    return jwt.decode(token, get_jwt_secret(), algorithms=[JWT_ALGORITHM])


def _extract_token(request: Request) -> str | None:
    token = request.cookies.get("access_token")
    if not token:
        auth = request.headers.get("Authorization", "")
        if auth.startswith("Bearer "):
            token = auth[7:]
    return token


async def get_current_user_factory(db):
    async def get_current_user(request: Request) -> dict:
        token = _extract_token(request)
        if not token:
            raise HTTPException(status_code=401, detail="Não autenticado")
        try:
            payload = decode_token(token)
            if payload.get("type") != "access":
                raise HTTPException(status_code=401, detail="Token inválido")
            user = await db.users.find_one({"_id": ObjectId(payload["sub"])})
            if not user:
                raise HTTPException(status_code=401, detail="Usuário não encontrado")
            user["id"] = str(user["_id"])
            user.pop("_id", None)
            user.pop("password_hash", None)
            return user
        except jwt.ExpiredSignatureError:
            raise HTTPException(status_code=401, detail="Sessão expirada")
        except jwt.InvalidTokenError:
            raise HTTPException(status_code=401, detail="Token inválido")

    return get_current_user


def require_admin(user: dict):
    if user.get("role") != "admin":
        raise HTTPException(status_code=403, detail="Acesso restrito ao administrador")

// webpack-health-plugin.js
// Webpack plugin that tracks compilation state and health metrics

class WebpackHealthPlugin {
  constructor() {
    this.status = {
      state: 'idle',           // idle, compiling, success, failed
      errors: [],
      warnings: [],
      lastCompileTime: null,
      lastSuccessTime: null,
      compileDuration: 0,
      totalCompiles: 0,
      firstCompileTime: null,
    };
  }

  apply(compiler) {
    const pluginName = 'WebpackHealthPlugin';

    // Hook: Compilation started
    compiler.hooks.compile.tap(pluginName, () => {
      const now = Date.now();
      this.status.state = 'compiling';
      this.status.lastCompileTime = now;

      if (!this.status.firstCompileTime) {
        this.status.firstCompileTime = now;
      }
    });

    // Hook: Compilation completed
    compiler.hooks.done.tap(pluginName, (stats) => {
      const info = stats.toJson({
        all: false,
        errors: true,
        warnings: true,
      });

      this.status.totalCompiles++;
      this.status.compileDuration = Date.now() - this.status.lastCompileTime;

      if (stats.hasErrors()) {
        this.status.state = 'failed';
        this.status.errors = info.errors.map(err => ({
          message: err.message || String(err),
          stack: err.stack,
          moduleName: err.moduleName,
          loc: err.loc,
        }));
      } else {
        this.status.state = 'success';
        this.status.lastSuccessTime = Date.now();
        this.status.errors = [];
      }

      if (stats.hasWarnings()) {
        this.status.warnings = info.warnings.map(warn => ({
          message: warn.message || String(warn),
          moduleName: warn.moduleName,
          loc: warn.loc,
        }));
      } else {
        this.status.warnings = [];
      }
    });

    // Hook: Compilation failed
    compiler.hooks.failed.tap(pluginName, (error) => {
      this.status.state = 'failed';
      this.status.errors = [{
        message: error.message,
        stack: error.stack,
      }];
      this.status.compileDuration = Date.now() - this.status.lastCompileTime;
    });

    // Hook: Invalid (file changed, recompiling)
    compiler.hooks.invalid.tap(pluginName, () => {
      this.status.state = 'compiling';
    });
  }

  getStatus() {
    return {
      ...this.status,
      // Add computed fields
      isHealthy: this.status.state === 'success',
      errorCount: this.status.errors.length,
      warningCount: this.status.warnings.length,
      hasCompiled: this.status.totalCompiles > 0,
    };
  }

  // Get simplified status for quick checks
  getSimpleStatus() {
    return {
      state: this.status.state,
      isHealthy: this.status.state === 'success',
      errorCount: this.status.errors.length,
      warningCount: this.status.warnings.length,
    };
  }

  // Reset statistics (useful for testing)
  reset() {
    this.status = {
      state: 'idle',
      errors: [],
      warnings: [],
      lastCompileTime: null,
      lastSuccessTime: null,
      compileDuration: 0,
      totalCompiles: 0,
      firstCompileTime: null,
    };
  }
}

module.exports = WebpackHealthPlugin;

/**
 * Format bytes to human-readable string
 * @param {number} bytes
 * @returns {string}
 */
function formatBytes(bytes) {
  if (bytes === 0) return '0 B';
  const k = 1024;
  const sizes = ['B', 'KB', 'MB', 'GB'];
  const i = Math.floor(Math.log(bytes) / Math.log(k));
  return Math.round(bytes / Math.pow(k, i) * 100) / 100 + ' ' + sizes[i];
}

/**
 * Format duration to human-readable string
 * @param {number} ms - Duration in milliseconds
 * @returns {string}
 */
function formatDuration(ms) {
  const seconds = Math.floor(ms / 1000);
  const minutes = Math.floor(seconds / 60);
  const hours = Math.floor(minutes / 60);

  if (hours > 0) {
    return `${hours}h ${minutes % 60}m ${seconds % 60}s`;
  } else if (minutes > 0) {
    return `${minutes}m ${seconds % 60}s`;
  } else {
    return `${seconds}s`;
  }
}

export function formatApiError(err) {
  const detail = err?.response?.data?.detail;
  if (!detail) return err?.message || "Erro inesperado";
  if (typeof detail === "string") return detail;
  if (Array.isArray(detail)) return detail.map((d) => d?.msg || JSON.stringify(d)).join(" ");
  return String(detail);
}

export function formatCurrency(v) {

import { createContext, useContext, useEffect, useState } from "react";
import { api } from "@/lib/api";

const AuthCtx = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null); // null = checking, false = anon, obj = user
  const [loading, setLoading] = useState(true);

  async function refresh() {
    try {
      const { data } = await api.get("/auth/me");
      setUser(data);
    } catch (e) {
      setUser(false);
    } finally {
      setLoading(false);
    }
  }

  useEffect(() => {
    refresh();
  }, []);

  async function login(email, password) {
    const { data } = await api.post("/auth/login", { email, password });
    if (data.token) localStorage.setItem("acorde_token", data.token);
    setUser(data.user);
    return data.user;
  }

  async function logout() {
    try { await api.post("/auth/logout"); } catch {}
    localStorage.removeItem("acorde_token");
    setUser(false);
  }

  return (
    <AuthCtx.Provider value={{ user, loading, login, logout, refresh }}>
      {children}
    </AuthCtx.Provider>
  );
}

export const useAuth = () => useContext(AuthCtx);

export default function Uploads() {
  const [uploads, setUploads] = useState([]);
  const [busy, setBusy] = useState(null);

  async function refresh() {
    const { data } = await api.get("/uploads");
    setUploads(data || []);
  }

  useEffect(() => { refresh(); }, []);

  async function handleUpload(type, file) {
    if (!file) return;
    setBusy(type);
    const fd = new FormData();
    fd.append("file", file);
    try {
      const { data } = await api.post(`/uploads/${type}`, fd, { headers: { "Content-Type": "multipart/form-data" } });
      toast.success(`${data.rows.toLocaleString("pt-BR")} linhas importadas (${data.competencia || "sem competência"})`);
      refresh();
    } catch (e) {
      toast.error(formatApiError(e));
    } finally {
      setBusy(null);
    }
  }

  async function removeUpload(id) {
    if (!window.confirm("Remover este upload e todas as linhas associadas?")) return;
    try {
      await api.delete(`/uploads/${id}`);
      toast.success("Upload removido");
      refresh();
    } catch (e) {
      toast.error(formatApiError(e));
    }
  }

  return (

        onDragLeave={() => setDrag(false)}
        onDrop={(e) => {

      >
        <UploadCloud className="w-7 h-7 text-neutral-500 mb-3" />

        <input
          ref={ref}
          type="file"
          accept={type.accept}
          className="hidden"
          onChange={(e) => onFile(e.target.files?.[0])}

  );
}

import { api, API, formatApiError, formatPct } from "@/lib/api";

import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter, DialogDescription } from "@/components/ui/dialog";
import { toast } from "sonner";
import { Pencil, Trash2, Plus, Upload, KeyRound, RefreshCw, FileDown, AlertCircle } from "lucide-react";

export default function Editoras() {
  const [editoras, setEditoras] = useState([]);
  const [filter, setFilter] = useState("all"); // all | pending | configured
  const [open, setOpen] = useState(false);
  const [editing, setEditing] = useState(null);
  const [form, setForm] = useState({ name: "", cnpj: "", admin_fee_percent: 0, email: "" });
  const [loginEditora, setLoginEditora] = useState(null);
  const [loginForm, setLoginForm] = useState({ email: "", password: "" });
  const bulkRef = useRef(null);
  const [syncing, setSyncing] = useState(false);

  async function refresh() {
    const { data } = await api.get("/editoras");
    setEditoras(data || []);
  }
  useEffect(() => { refresh(); }, []);

  async function syncFromReports() {
    setSyncing(true);
    try {
      const { data } = await api.post("/discover/sync");
      toast.success(`${data.new_editoras} editoras e ${data.new_autores} autores importados dos relatórios`);
      refresh();
    } catch (e) { toast.error(formatApiError(e)); } finally { setSyncing(false); }
  }

  async function downloadPendingCsv() {
    try {
      const res = await api.get("/discover/pending/editoras.csv", { responseType: "blob" });
      const a = document.createElement("a");
      a.href = URL.createObjectURL(new Blob([res.data], { type: "text/csv" }));
      a.download = "acorde_editoras_pendentes.csv";
      document.body.appendChild(a);
      a.click();
      a.remove();
      URL.revokeObjectURL(a.href);
      toast.success("CSV pendentes baixado");
    } catch (e) { toast.error(formatApiError(e)); }
  }

  const visible = editoras.filter((e) => {
    if (filter === "pending") return e.needs_review;
    if (filter === "configured") return !e.needs_review;
    return true;
  });

  const pendingCount = editoras.filter((e) => e.needs_review).length;

  function openNew() {
    setEditing(null);
    setForm({ name: "", cnpj: "", admin_fee_percent: 0, email: "" });
    setOpen(true);
  }

  function openEdit(e) {
    setEditing(e);
    setForm({ name: e.name, cnpj: e.cnpj || "", admin_fee_percent: e.admin_fee_percent || 0, email: e.email || "" });
    setOpen(true);
  }

  async function save() {
    try {
      if (editing) {
        await api.patch(`/editoras/${editing.id}`, {
          ...form,
          admin_fee_percent: Number(form.admin_fee_percent || 0),
        });
        toast.success("Editora atualizada");
      } else {
        await api.post("/editoras", { ...form, admin_fee_percent: Number(form.admin_fee_percent || 0) });
        toast.success("Editora cadastrada");
      }
      setOpen(false);
      refresh();
    } catch (e) { toast.error(formatApiError(e)); }
  }

  async function remove(id) {
    if (!window.confirm("Remover esta editora?")) return;
    try { await api.delete(`/editoras/${id}`); refresh(); } catch (e) { toast.error(formatApiError(e)); }
  }

  async function bulkCsv(file) {
    if (!file) return;
    const fd = new FormData();
    fd.append("file", file);
    try {
      const { data } = await api.post("/editoras/bulk", fd, { headers: { "Content-Type": "multipart/form-data" } });
      toast.success(`${data.created} criadas, ${data.updated} atualizadas`);
      refresh();
    } catch (e) { toast.error(formatApiError(e)); }
  }

  async function createLogin() {
    try {
      await api.post(`/editoras/${loginEditora.id}/login`, {
        editora_id: loginEditora.id,
        email: loginForm.email,
        password: loginForm.password,
      });
      toast.success("Login criado para a editora");
      setLoginEditora(null);
      setLoginForm({ email: "", password: "" });
      refresh();
    } catch (e) { toast.error(formatApiError(e)); }
  }

  return (
    <div className="space-y-10">
      <div className="flex items-start justify-between flex-wrap gap-4">
        <div>
          <div className="text-xs uppercase tracking-[0.3em] text-[#FFD700] mb-2">Clientes</div>
          <h1 className="text-5xl font-extrabold leading-none">Editoras</h1>
          <p className="text-neutral-500 mt-3 text-sm max-w-2xl">
            Cadastre as editoras (clientes Acorde) e defina a taxa administrativa cobrada por cada uma.
            Editoras encontradas nos relatórios Abramus são importadas automaticamente como <span className="text-[#FFD700]">pendentes</span>.
          </p>
        </div>
        <div className="flex items-center gap-2 flex-wrap">
          <Button data-testid="sync-reports-btn" variant="outline" disabled={syncing} onClick={syncFromReports} className="h-11 border-[#2a2a2a] rounded-sm hover:border-[#FFD700] hover:text-[#FFD700]">
            <RefreshCw className={`w-4 h-4 mr-2 ${syncing ? "animate-spin" : ""}`} /> Sincronizar dos relatórios
          </Button>
          <Button data-testid="download-pending-editoras" variant="outline" onClick={downloadPendingCsv} className="h-11 border-[#2a2a2a] rounded-sm hover:border-[#FFD700] hover:text-[#FFD700]">

"use client";
// Inspired by react-hot-toast library
import * as React from "react"

const TOAST_LIMIT = 1
const TOAST_REMOVE_DELAY = 1000000

const actionTypes = {
  ADD_TOAST: "ADD_TOAST",
  UPDATE_TOAST: "UPDATE_TOAST",
  DISMISS_TOAST: "DISMISS_TOAST",
  REMOVE_TOAST: "REMOVE_TOAST"
}

let count = 0

function genId() {
  count = (count + 1) % Number.MAX_SAFE_INTEGER
  return count.toString();
}

const toastTimeouts = new Map()

const addToRemoveQueue = (toastId) => {
  if (toastTimeouts.has(toastId)) {
    return
  }

  const timeout = setTimeout(() => {
    toastTimeouts.delete(toastId)
    dispatch({
      type: "REMOVE_TOAST",
      toastId: toastId,
    })
  }, TOAST_REMOVE_DELAY)

  toastTimeouts.set(toastId, timeout)
}

export const reducer = (state, action) => {
  switch (action.type) {
    case "ADD_TOAST":
      return {
        ...state,
        toasts: [action.toast, ...state.toasts].slice(0, TOAST_LIMIT),
      };

    case "UPDATE_TOAST":
      return {
        ...state,
        toasts: state.toasts.map((t) =>
          t.id === action.toast.id ? { ...t, ...action.toast } : t),
      };

    case "DISMISS_TOAST": {
      const { toastId } = action

      // ! Side effects ! - This could be extracted into a dismissToast() action,
      // but I'll keep it here for simplicity
      if (toastId) {
        addToRemoveQueue(toastId)
      } else {
        state.toasts.forEach((toast) => {
          addToRemoveQueue(toast.id)
        })
      }

      return {
        ...state,
        toasts: state.toasts.map((t) =>
          t.id === toastId || toastId === undefined
            ? {
                ...t,
                open: false,
              }
            : t),
      };
    }
    case "REMOVE_TOAST":
      if (action.toastId === undefined) {
        return {
          ...state,
          toasts: [],
        }
      }
      return {
        ...state,
        toasts: state.toasts.filter((t) => t.id !== action.toastId),
      };
  }
}

const listeners = []

let memoryState = { toasts: [] }

function dispatch(action) {
  memoryState = reducer(memoryState, action)
  listeners.forEach((listener) => {
    listener(memoryState)
  })
}

function toast({
  ...props
}) {
  const id = genId()

  const update = (props) =>
    dispatch({
      type: "UPDATE_TOAST",
      toast: { ...props, id },
    })
  const dismiss = () => dispatch({ type: "DISMISS_TOAST", toastId: id })

  dispatch({
    type: "ADD_TOAST",
    toast: {
      ...props,
      id,
      open: true,
      onOpenChange: (open) => {
        if (!open) dismiss()
      },
    },
  })

  return {
    id: id,
    dismiss,
    update,
  }
}

function useToast() {
  const [state, setState] = React.useState(memoryState)

  React.useEffect(() => {
    listeners.push(setState)
    return () => {
      const index = listeners.indexOf(setState)
      if (index > -1) {
        listeners.splice(index, 1)
      }
    };
  }, [state])

  return {
    ...state,
    toast,
    dismiss: (toastId) => dispatch({ type: "DISMISS_TOAST", toastId }),
  };
}

export { useToast, toast }

const path = require("path");
require("dotenv").config();

// Check if we're in development/preview mode (not production build)
// Craco sets NODE_ENV=development for start, NODE_ENV=production for build
const isDevServer = process.env.NODE_ENV !== "production";

// Environment variable overrides
const config = {
  enableHealthCheck: process.env.ENABLE_HEALTH_CHECK === "true",
};

function makeDevServerV5Compatible(devServerConfig) {
  const {
    https,
    onAfterSetupMiddleware,
    onBeforeSetupMiddleware,
    onListening,
    setupMiddlewares,
    ...compatibleConfig
  } = devServerConfig;

  compatibleConfig.server =
    typeof https === "object"
      ? { type: "https", options: https }
      : https
        ? "https"
        : "http";
  compatibleConfig.headers = {
    ...compatibleConfig.headers,
    "Cross-Origin-Resource-Policy": "same-origin",
  };

  if (onBeforeSetupMiddleware || setupMiddlewares) {
    compatibleConfig.setupMiddlewares = (middlewares, devServer) => {
      if (onBeforeSetupMiddleware) {
        onBeforeSetupMiddleware(devServer);
      }

      return setupMiddlewares
        ? setupMiddlewares(middlewares, devServer)
        : middlewares;
    };
  }

  compatibleConfig.onListening = (devServer) => {
    devServer.close ??= (callback) => devServer.stopCallback(callback);

    if (onListening) {
      onListening(devServer);
    }
    if (onAfterSetupMiddleware) {
      onAfterSetupMiddleware(devServer);
    }
  };

  return compatibleConfig;
}

  let overlay = emergentOverlay;

      if (previousSetupMiddlewares) {
        middlewares = previousSetupMiddlewares(middlewares, devServer);
      }

      return middlewares;
    };

    return devServerConfig;
  };
}

# Acorde — Royalties Processor (PRD)

## Problem Statement (original, Portuguese)

Site de processamento e conversão de arquivos para a Acorde — administradora
de direitos autorais musicais. Clientes = editoras; editoras têm autores. A
Acorde cobra uma taxa administrativa % individualizada por editora e, quando
cabe, também do autor. A fonte dos dados são 3 relatórios exportados da
Abramus Digital:

- Relatório Geral (ZIP de CSVs ou CSV único): recebimento total consolidado
  por editora.
- Relatório Editora (XLSX): participação da editora no contrato de edição
  por obra.
- Relatório Autor (XLSX): valor bruto por autor, obra, fonte, território...

O sistema deve calcular a taxa administrativa (fonte: banco cadastrado via
UI ou CSV em massa), gerar relatórios Acorde detalhados por editora e por
autor, exportar PDFs com marca Acorde (preto/amarelo/branco) — e permitir
login individual para cada editora.

## Users
- Admin Acorde (aline@acordemusica.com.br): tudo.
- Editora: vê somente a si mesma, seus autores e seus relatórios.

## Core Requirements (static)
- Upload dos 3 tipos de relatório Abramus; parsing robusto BR (vírgula
  decimal, latin-1 fallback, ZIP com múltiplos CSVs).
- CRUD Editoras (nome, CNPJ, taxa_admin %, email, login opcional).
- CRUD Autores (nome, editora vinculada, taxa_admin %).
- Importação em massa via CSV para editoras e autores.
- Relatório editora com cálculo da taxa admin (bruto, taxa, líquido).
- Relatório autor que NÃO exibe a taxa admin — somente o valor líquido final.
- Exportação PDF com cabeçalho e rodapé Acorde (preto/amarelo).
- Dashboard admin: KPIs e gráfico por fonte (Spotify, YouTube...).
- Dashboard editora: escopo restrito.

## Architecture
- Backend: FastAPI, Motor (MongoDB async), JWT + bcrypt auth, pandas +
  openpyxl para XLSX, reportlab para PDFs.
- Frontend: React 19, Shadcn UI, Tailwind, Recharts, Sonner.
- MongoDB coleções: users, editoras, autores, uploads, royalty_rows.

## What's been implemented (Feb 2026)
- JWT auth (admin + editora), admin seeding idempotente a partir de .env.
- Upload de .zip/.csv/.xlsx para os 3 tipos de relatório Abramus.
- Parsers normalizam valores BR e armazenam linhas em `royalty_rows` com
  `titular_norm`/`nome_norm` para casamento.
- CRUD completo + importação CSV para editoras e autores (com e-mail).
- Criação de login para editora (gera user role=editora com editora_id).
- Dashboard admin com KPIs (valor total geral/editora/autor) + chart por
  fonte + top editoras.
- Dashboard editora com escopo restrito.
- Visualização e PDF de relatório Editora (com taxa admin).
